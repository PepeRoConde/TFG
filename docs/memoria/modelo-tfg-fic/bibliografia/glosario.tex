\newglossaryentry{interpretabilidade}{
  name=interpretabilidade,
	description={Cualidade intrínsica dun sistema de toma de decisións que permite coñecelo, de xeito que os seus compoñentes están xustificados e enténdese o seu rol}
}
\newglossaryentry{explicabilidade}{
  name=explicabilidade,
  description={Cualidade dun sistema de toma de decisións xunto con ferramentas auxiliares ou non de xustificar as súas decisións. Responde á pregunta: Porqué se tomou esta decisión?}
}

  name=token,
	description={Vector que apréndese que forma parte da entrada e fluxo interno de modelos Transformadores.}% (e.g. ViT (ver Sección \ref{sec:vit}), \textsc{Crate} \ref{sec:crate}).}
}
