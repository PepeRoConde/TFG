\begin{frame}{O \textsc{Scrr} nos Transformers}
				Os tokens son unha combinación de espazos ($z_i \doteq \alpha_i U_{[K]}$)

				\begin{figure}[ht]
        \centering
        \includegraphics[width=\linewidth]{imaxes/scrr.drawio.pdf}
    		\caption*{}
\end{figure}

\end{frame}

\begin{frame}{O \textsc{Scrr} nos Transformers}
\begin{block}{}
defínese como a maximización da seguinte función: \[\Delta R (Z | U_{[K]}) = R(Z) - R^{c}(Z | U_{[K]}) - ||Z||_0\] onde \[R(Z) \dot{=} \frac{1}{2} \text{logdet}(I+ Z^{'}Z)\] e \[R^{c}(Z | U_{[K]}) \dot{=} \sum_{k}R(U^{'}_{k}Z)\]\end{block}
\end{frame}

\section{Discusión de Resultados}

\begin{frame}{Segunda orde}
    \begin{block}{}
        \begin{itemize}
            \item A segunda cabeza imita á cabeza orixinal \pause
            \item A primeira pasa a ser unha versión suavizada \pause
            \item Só supera ó orixinal na tarefa sinxela (DRIVE)
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Cruzamento de ordes}
    \begin{block}{}
        \begin{itemize}
            \item Precisión $\approx$ azar (0.1\,\%) en ambos sentidos \pause
            \item Consecuencia do novo rol das cabezas
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Compartir $U$ e $D$}
    \begin{block}{}
        \begin{itemize}
            \item 8M $\rightarrow$ 6.5M parámetros \pause
            \item Rendemento comparable (RFMiD: AUC 0.877 vs. 0.878) \pause
            \item Atención máis variada
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{MCR$^2$ e dispersión}
    \begin{block}{}
        \begin{itemize}
            \item As dúas métricas compiten \pause
            \item Ningunha variante domina á resto \pause
            \item Combo $>$ orixinal en DRIVE
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Explicabilidade}
    \begin{block}{}
        \begin{itemize}
            \item Mapas de atención: separación semántica e utilidade clínica \pause
            \item Dimensións: diagnóstico da propia rede \pause
            \item RFMiD: a variante que comparte explica máis
        \end{itemize}
    \end{block}
\end{frame}
