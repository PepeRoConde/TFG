\begin{frame}{Unha posible solución}
				\begin{block}{Teoría do \textit{Sparse Coding Rate Reduction} (dende 2007)}
	Un obxectivo de optimización que busca representacións: \pause
    \begin{itemize}
						\item \textcolor{udcpink}{Lineais:} expresarse como combinación lineal de conceptos  \pause
						\item \textcolor{udcpink}{Comprimidas:} que cada concepto ``ocupe'' o menor espazo\pause
						\item \textcolor{udcpink}{Dispersas:} expresarse como combinación de ``poucos'' conceptos
    \end{itemize}
\end{block}\pause
			\centering	\fbox{ ``Que'' se expresa? En que Rede Neuronal convértese todo isto?}
\end{frame}

\begin{frame}{O \textsc{Scrr} nos Transformers}
				Exprésanse \textcolor{udcpink}{tokens} como  unha combinación de espazos ($z_i \doteq \alpha_i U_{[K]}$). Todos os tokens da capa $\ell$ son $Z^\ell$

				\begin{figure}[ht]
        \centering
        \includegraphics[width=\linewidth]{imaxes/presentacion/scrr.drawio.pdf}
    		\caption*{}
\end{figure}

\end{frame}

\begin{frame}{O \textsc{Scrr} nos Transformers}
\begin{block}{}
				Defínese como a minimización da seguinte función: \[\Delta R (Z | U_{[K]}) =  \underbrace{\|Z\|_0}_{\text{Dispersión}} +\underbrace{R^{c}(Z | U_{[K]})}_{\text{Compresión}} - \underbrace{R(Z)}_{\text{Expansión}}\] onde $\|Z\|_0$ e cantos ceros hai, \[R(Z) \doteq \frac{1}{2} \operatorname{logdet}(I+ Z^{'}Z),\] e \[R^{c}(Z | U_{[K]}) \doteq \sum_{k}R(U^{'}_{k}Z)\]\end{block}\pause

			\centering	\fbox{ Como convirtése isto nun Transformer?}
\end{frame}

\begin{frame}{A arquitectura \textsc{Crate}}

				\begin{block}{Optimizando o obxectivo \textsc{Scrr} con \textit{Algorithm Unrolling}}
\begin{itemize}
				\item \textcolor{udcpink}{Multicabeza de Atención ($U$):} implementa a ``Compresión''
				\item \textcolor{udcpink}{MLP ($D$):} implementa a ``Dispersión'' e a ``Expansión''
\end{itemize}
				\end{block}\pause

				\begin{block}{Propiedades}
\begin{itemize}
				\item \textit{Interpretabilidade}, por construción
				\item Cabezas de Atención ``ortogonais'' $\rightarrow$ \textit{explicabilidade}
\end{itemize}
				\end{block}

\section{Discusión de Resultados}

\begin{frame}{Segunda orde}
    \begin{block}{}
        \begin{itemize}
            \item A segunda cabeza imita á cabeza orixinal \pause
            \item A primeira pasa a ser unha versión suavizada \pause
            \item Só supera ó orixinal na tarefa sinxela (DRIVE)
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Cruzamento de ordes}
    \begin{block}{}
        \begin{itemize}
            \item Precisión $\approx$ azar (0.1\,\%) en ambos sentidos \pause
            \item Consecuencia do novo rol das cabezas
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Compartir $U$ e $D$}
    \begin{block}{}
        \begin{itemize}
            \item 8M $\rightarrow$ 6.5M parámetros \pause
            \item Rendemento comparable (RFMiD: AUC 0.877 vs. 0.878) \pause
            \item Atención máis variada
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{CR$^2$ e dispersión}
    \begin{block}{}
        \begin{itemize}
            \item As dúas métricas compiten \pause
            \item Ningunha variante domina á resto \pause
            \item Combo $>$ orixinal en DRIVE
        \end{itemize}
    \end{block}
\end{frame}
